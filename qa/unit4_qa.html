Q1 Command Line Arguments: Write a program that prints a table indicating the number of occurrences of each alphabet in a text entered specifically as command line arguments.
Example Program
// Count occurrences of each alphabet from command line arguments
#include <stdio.h>
#include <ctype.h>

int main(int argc, char *argv[]) {
    int count[26] = {0};
    int i, j;

    if (argc < 2) {
        printf("Usage: %s <text>\n", argv[0]);
        return 1;
    }

    // Process each argument, character by character
    for (i = 1; i < argc; i++) {
        for (j = 0; argv[i][j] != '\0'; j++) {
            if (isalpha(argv[i][j])) {
                char ch = tolower(argv[i][j]);
                count[ch - 'a']++;
            }
        }
    }

    printf("Letter | Count\n");
    printf("-------+------\n");
    for (i = 0; i < 26; i++) {
        if (count[i] > 0)
            printf("   %c   |  %d\n", 'a' + i, count[i]);
    }

    return 0;
}

Q2 File Management (Copy): Write a program to copy the content of one file to another.
Example Program
// Copy the content of one file to another
#include <stdio.h>
#include <stdlib.h>

int main() {
    FILE *src, *dest;
    char ch;

    src = fopen("source.txt", "r");
    if (src == NULL) {
        printf("Cannot open source file\n");
        exit(1);
    }

    dest = fopen("destination.txt", "w");
    if (dest == NULL) {
        printf("Cannot open destination file\n");
        fclose(src);
        exit(1);
    }

    // Read and write one character at a time
    while ((ch = fgetc(src)) != EOF) {
        fputc(ch, dest);
    }

    printf("File copied successfully\n");

    fclose(src);
    fclose(dest);

    return 0;
}
Output
File copied successfully

Q6 Structures and Functions: Write a program to pass a structure variable to a function to calculate the total salary of an employee.
Example Program
// Pass a structure to a function to compute total salary
#include <stdio.h>

struct Employee {
    char name[30];
    float basic;
    float hra;
    float da;
};

// Structure passed by value
float calculateSalary(struct Employee e) {
    return e.basic + e.hra + e.da;
}

int main() {
    struct Employee emp = {"Priya Nair", 30000.0, 8000.0, 5000.0};

    float total = calculateSalary(emp);

    printf("Employee: %s\n", emp.name);
    printf("Total Salary: %.2f\n", total);

    return 0;
}
Output
Employee: Priya Nair
Total Salary: 43000.00

Q9 File Opening Modes: Write a program to demonstrate different file opening modes such as read (r), write (w), and append (a) to modify text files.
Example Program
// Demonstrate "w", "a", and "r" file modes
#include <stdio.h>

int main() {
    FILE *fp;
    char line[100];

    // "w" mode: creates the file, or overwrites it if it exists
    fp = fopen("notes.txt", "w");
    fprintf(fp, "First line.\n");
    fclose(fp);

    // "a" mode: appends without erasing existing content
    fp = fopen("notes.txt", "a");
    fprintf(fp, "Second line (appended).\n");
    fclose(fp);

    // "r" mode: reads existing content
    fp = fopen("notes.txt", "r");
    printf("File contents:\n");
    while (fgets(line, sizeof(line), fp) != NULL) {
        printf("%s", line);
    }
    fclose(fp);

    return 0;
}
Output
File contents:
First line.
Second line (appended).
Q10 Sequential and Random Access: Write a program to demonstrate random access in a file using operations like fseek() or ftell() to read data from a specific position.
Example Program
// Random access using fseek() and ftell()
#include <stdio.h>

int main() {
    FILE *fp;
    char ch;

    // Write a known string to a file first
    fp = fopen("data.txt", "w");
    fprintf(fp, "ABCDEFGHIJ");
    fclose(fp);

    fp = fopen("data.txt", "r");

    // Jump directly to the 5th byte (index 4) from the start
    fseek(fp, 4, SEEK_SET);
    ch = fgetc(fp);
    printf("Character at position 4: %c\n", ch);

    // ftell() reports the current position in the file
    printf("Current position: %ld\n", ftell(fp));

    // Move back 2 bytes from the current position
    fseek(fp, -2, SEEK_CUR);
    ch = fgetc(fp);
    printf("Character after seeking back 2: %c\n", ch);

    fclose(fp);

    return 0;
}
Output
Character at position 4: E
Current position: 5
Character after seeking back 2: D
Q11 Self-referential Structures: Write a program to declare a self-referential structure, typically used for creating linked lists.
Example Program
// Self-referential structure forming a simple linked list
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;  // pointer to a structure of the same type
};

int main() {
    // Manually build a list of 3 nodes: 10 -> 20 -> 30
    struct Node *head = (struct Node *)malloc(sizeof(struct Node));
    struct Node *second = (struct Node *)malloc(sizeof(struct Node));
    struct Node *third = (struct Node *)malloc(sizeof(struct Node));

    head->data = 10;
    head->next = second;

    second->data = 20;
    second->next = third;

    third->data = 30;
    third->next = NULL;

    // Traverse and print the list
    struct Node *temp = head;
    while (temp != NULL) {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }
    printf("NULL\n");

    free(head);
    free(second);
    free(third);

    return 0;
}
Output
10 -> 20 -> 30 -> NULL
Q12 Bit-fields: Write a program to define a structure with bit-fields to demonstrate efficient memory usage for storing status flags.
Example Program
// Bit-fields for compact status flag storage
#include <stdio.h>

struct StatusFlags {
    unsigned int isActive   : 1;
    unsigned int isAdmin    : 1;
    unsigned int isVerified : 1;
    unsigned int priority   : 4;  // values 0-15
};

int main() {
    struct StatusFlags user;

    user.isActive   = 1;
    user.isAdmin    = 0;
    user.isVerified = 1;
    user.priority   = 9;

    printf("Active: %u\n", user.isActive);
    printf("Admin: %u\n", user.isAdmin);
    printf("Verified: %u\n", user.isVerified);
    printf("Priority: %u\n", user.priority);

    // All four fields fit inside a single int-sized block
    printf("Size of struct: %zu bytes\n", sizeof(struct StatusFlags));

    return 0;
}
Output
Active: 1
Admin: 0
Verified: 1
Priority: 9
Size of struct: 4 bytes
