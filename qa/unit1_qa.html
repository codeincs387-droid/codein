Q1 Swap Two Numbers: Write a program to swap two numbers using a temporary variable.
Example Program
// Swap two numbers using a temporary variable
#include <stdio.h>

int main() {
    int a = 5, b = 10, temp;

    temp = a;
    a = b;
    b = temp;

    printf("a = %d, b = %d\n", a, b);

    return 0;
}
Output
a = 10, b = 5
Q2 Basic Input/Output: Write a program to read an integer, a floating-point number, and a character from the user and display them using scanf and printf.
Example Program
// Read and display an int, a float, and a char
#include <stdio.h>

int main() {
    int i;
    float f;
    char c;

    printf("Enter an integer: ");
    scanf("%d", &i);

    printf("Enter a float: ");
    scanf("%f", &f);

    printf("Enter a character: ");
    scanf(" %c", &c);

    printf("\nYou entered:\n");
    printf("Integer: %d\n", i);
    printf("Float: %.2f\n", f);
    printf("Character: %c\n", c);

    return 0;
}
Output
Enter an integer: 25
Enter a float: 3.14
Enter a character: A

You entered:
Integer: 25
Float: 3.14
Character: A

Q7 Increment and Decrement: Write a program to demonstrate the difference between prefix and postfix increment (++) and decrement (--) operators.
Example Program
// Prefix vs postfix increment and decrement
#include <stdio.h>

int main() {
    int a = 5;

    printf("a++ (postfix) gives: %d\n", a++);
    printf("a is now: %d\n", a);

    printf("++a (prefix) gives: %d\n", ++a);
    printf("a is now: %d\n", a);

    int b = 5;

    printf("b-- (postfix) gives: %d\n", b--);
    printf("--b (prefix) gives: %d\n", --b);

    return 0;
}
Output
a++ (postfix) gives: 5
a is now: 6
++a (prefix) gives: 7
a is now: 7
b-- (postfix) gives: 5
--b (prefix) gives: 3
Q8 Logical Operators: Write a program that uses logical operators (&&, ||, !) to check if a number is both positive and even.
Example Program
// Check if a number is both positive and even
#include <stdio.h>

int main() {
    int n = 14;

    if (n > 0 && n % 2 == 0)
        printf("%d is positive and even\n", n);
    else
        printf("%d is not positive and even\n", n);

    if (!(n < 0))
        printf("%d is not negative\n", n);

    return 0;
}
Output
14 is positive and even
14 is not negative

Q10 Operator Precedence: Write a program to evaluate a complex expression (e.g., a + b * c / d) and print the result to demonstrate the order of precedence and associativity.
Example Program
// Operator precedence: a + b * c / d
#include <stdio.h>

int main() {
    int a = 5, b = 4, c = 3, d = 2, result;

    // * and / have higher precedence than +, and are left-associative
    result = a + b * c / d;

    printf("a + b * c / d = %d\n", result);
    // Evaluated as: a + ((b * c) / d) = 5 + (12 / 2) = 5 + 6 = 11

    return 0;
}
Output
a + b * c / d = 11
