A1 Array Operations: A menu-driven program to print even/odd elements, calculate sum/average, find max/min, remove duplicates, and reverse an array.
Example Program
// Menu-driven array operations
#include <stdio.h>

void printEvenOdd(int arr[], int n) {
    printf("Even elements: ");
    for (int i = 0; i < n; i++)
        if (arr[i] % 2 == 0) printf("%d ", arr[i]);
    printf("\nOdd elements: ");
    for (int i = 0; i < n; i++)
        if (arr[i] % 2 != 0) printf("%d ", arr[i]);
    printf("\n");
}

void sumAverage(int arr[], int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) sum += arr[i];
    printf("Sum: %d, Average: %.2f\n", sum, (float)sum / n);
}

void maxMin(int arr[], int n) {
    int max = arr[0], min = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > max) max = arr[i];
        if (arr[i] < min) min = arr[i];
    }
    printf("Max: %d, Min: %d\n", max, min);
}

void removeDuplicates(int arr[], int n) {
    printf("Unique elements: ");
    for (int i = 0; i < n; i++) {
        int isDup = 0;
        for (int j = 0; j < i; j++)
            if (arr[i] == arr[j]) isDup = 1;
        if (!isDup) printf("%d ", arr[i]);
    }
    printf("\n");
}

void reverseArray(int arr[], int n) {
    printf("Reversed array: ");
    for (int i = n - 1; i >= 0; i--)
        printf("%d ", arr[i]);
    printf("\n");
}

int main() {
    int arr[] = {4, 8, 15, 8, 16, 23, 4, 42};
    int n = 8, choice;

    printf("1. Even/Odd  2. Sum/Average  3. Max/Min  4. Remove Duplicates  5. Reverse\n");
    printf("Enter choice: ");
    scanf("%d", &choice);

    switch (choice) {
        case 1: printEvenOdd(arr, n); break;
        case 2: sumAverage(arr, n); break;
        case 3: maxMin(arr, n); break;
        case 4: removeDuplicates(arr, n); break;
        case 5: reverseArray(arr, n); break;
        default: printf("Invalid choice\n");
    }
    return 0;
}
Output
1. Even/Odd  2. Sum/Average  3. Max/Min  4. Remove Duplicates  5. Reverse
Enter choice: 4
Unique elements: 4 8 15 16 23 42
A2 Array Merging: Merge two ordered arrays into one ordered array.
Example Program
// Merge two ordered arrays
#include <stdio.h>

int main() {
    int a[] = {1, 3, 5, 7};
    int b[] = {2, 4, 6, 8, 10};
    int n1 = 4, n2 = 5;
    int merged[9];
    int i = 0, j = 0, k = 0;

    while (i < n1 && j < n2) {
        if (a[i] <= b[j])
            merged[k++] = a[i++];
        else
            merged[k++] = b[j++];
    }
    while (i < n1) merged[k++] = a[i++];
    while (j < n2) merged[k++] = b[j++];

    printf("Merged array: ");
    for (i = 0; i < n1 + n2; i++)
        printf("%d ", merged[i]);
    printf("\n");

    return 0;
}
Output
Merged array: 1 2 3 4 5 6 7 8 10

A4 String Operations: A menu-driven program for showing the address of each character, concatenating strings without strcat, comparing two strings, converting case, calculating the number of vowels, and reversing a string.
Example Program
// Menu-driven string operations
#include <stdio.h>
#include <string.h>
#include <ctype.h>

void showAddresses(char str[]) {
    for (int i = 0; str[i] != '\0'; i++)
        printf("Address of '%c': %p\n", str[i], (void*)&str[i]);
}

void concatWithoutStrcat(char s1[], char s2[]) {
    int i = 0, j = 0;
    while (s1[i] != '\0') i++;
    while (s2[j] != '\0') {
        s1[i] = s2[j];
        i++; j++;
    }
    s1[i] = '\0';
    printf("Concatenated string: %s\n", s1);
}

void compareStrings(char s1[], char s2[]) {
    int result = strcmp(s1, s2);
    if (result == 0)
        printf("Strings are equal\n");
    else if (result < 0)
        printf("\"%s\" comes before \"%s\"\n", s1, s2);
    else
        printf("\"%s\" comes after \"%s\"\n", s1, s2);
}

void convertCase(char str[]) {
    printf("Converted: ");
    for (int i = 0; str[i] != '\0'; i++) {
        if (islower(str[i]))
            printf("%c", toupper(str[i]));
        else
            printf("%c", tolower(str[i]));
    }
    printf("\n");
}

void countVowels(char str[]) {
    int count = 0;
    for (int i = 0; str[i] != '\0'; i++) {
        char c = tolower(str[i]);
        if (c=='a'||c=='e'||c=='i'||c=='o'||c=='u')
            count++;
    }
    printf("Number of vowels: %d\n", count);
}

void reverseString(char str[]) {
    int len = strlen(str);
    for (int i = len - 1; i >= 0; i--)
        printf("%c", str[i]);
    printf("\n");
}

int main() {
    char str[50] = "Hello";
    int choice;

    printf("1. Char Addresses  2. Concatenate  3. Compare  4. Convert Case  5. Count Vowels  6. Reverse\n");
    printf("Enter choice: ");
    scanf("%d", &choice);

    switch (choice) {
        case 1: showAddresses(str); break;
        case 2: {
            char s2[] = " World";
            concatWithoutStrcat(str, s2);
            break;
        }
        case 3: {
            char s2[] = "Hero";
            compareStrings(str, s2);
            break;
        }
        case 4: convertCase(str); break;
        case 5: countVowels(str); break;
        case 6: printf("Reversed: "); reverseString(str); break;
        default: printf("Invalid choice\n");
    }
    return 0;
}

Reversed: olleH
